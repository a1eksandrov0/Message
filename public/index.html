function showAuthModal() { document.getElementById('auth-modal').classList.remove('hidden'); clearError(); } function hideAuthModal() { document.getElementById('auth-modal').classList.add('hidden'); } function validateEnglishPassword(password) { return /^[A-Za-z0-9!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]+$/.test(password); } // Создание аккаунта при нажатии на ссылку "Создание Аккаунта Message" async function register() { clearError(); const password = document.getElementById('auth-password').value.trim(); if (!password) { return showError('Задайте пароль перед созданием аккаунта'); } if (!validateEnglishPassword(password)) { return showError('Пароль должен содержать только английские буквы и символы'); } try { const res = await fetch('/api/register', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ password }) }); const data = await res.json(); if (data.error) return showError(data.error); myId = data.id; alert(`Ваш новый ID Message: ${myId}`); await openChatScreen(); } catch (e) { showError('Ошибка соединения с сервером'); } } // Вход по кнопке "Продолжить" async function login() { clearError(); const id = document.getElementById('auth-id').value.trim(); const password = document.getElementById('auth-password').value.trim(); if (!id || !password) { return showError('Введите ваш ID и пароль'); } try { const res = await fetch('/api/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id, password }) }); const data = await res.json(); if (data.error) return showError(data.error); myId = data.id; await openChatScreen(); } catch (e) { showError('Ошибка соединения с сервером'); } }
